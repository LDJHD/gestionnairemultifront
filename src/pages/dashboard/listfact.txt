import axios from "axios";
import { useEffect,useState } from "react";

const FacturePreview = ({ formData,selectedProducts, onClose }) => {

const [clients,setclients]=useState([]);
    const calculateTotal = () => {
      return selectedProducts.reduce((acc, product) => acc + product.prix * product.quantity, 0);
    };

      // Fonction pour récupérer les détails du produit
      const fetchclientsDetails = async (id) => {
        try {
            const token = localStorage.getItem("token"); // Assurez-vous que le jeton est bien stocké
    
            const response = await axios.post(
                "${import.meta.env.VITE_API_URL}/api/v1/client/detailById",
                { id },
                {
                    headers: {
                        Authorization: `Bearer ${token}`, // Transmettre le jeton dans les en-têtes
                    },
                }
            );
            
            setclients( response.data);
            console.log('fffffff'+clients);
    
        
        } catch (error) {
            console.error("Erreur lors de la récupération des détails du client :", error);
        }
    };
    
      
        useEffect(() => {
          if (formData.client) {
            fetchclientsDetails(formData.client);
            console.log(formData.client)
             // Charger les détails si l'ID du client est fourni
          }else {
            console.log('loose');
          }
        }, [formData.client]);
  
  
  
    return (
      <div className="fixed top-0 z-50 left-0 w-full h-full bg-black bg-opacity-50 flex justify-center items-center">
        <div className="bg-white p-6 rounded-lg shadow-lg w-1/2">
          <h2 className="text-xl font-bold mb-4">Prévisualisation de la Facture</h2>
          <p>Client: {clients.nom} {clients.prenom}</p>
          <p>Remise: {formData.remise} €</p>
          <p>IFU: {clients.ifu}</p>
          <table className="w-full mt-4 text-justify">
            <thead>
              <tr>
                <th>Libelle</th>
                <th>Prix</th>
                <th>Quantité</th>
                <th>Total</th>
              </tr>
            </thead>
            <tbody>
              {selectedProducts.map((product) => (
                <tr key={product.code_barre}>
                  <td>{product.nom}</td>
                  <td>{product.prix} Fcfa</td>
                  <td>{product.quantity} {product.unit}</td>
                  <td>{product.prix * product.quantity} €</td>
                </tr>
              ))}
            </tbody>
          </table>
          <p className="mt-4 font-bold">Total: {calculateTotal()} €</p>
          <button onClick={onClose} className="mt-4 p-2 bg-blue-600 text-white rounded">
            Fermer
          </button>
        </div>
      </div>
    );
  };
  
  export default FacturePreview;
